cmake_minimum_required(VERSION 3.16)
project(checkers_cpp VERSION 1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/Project-Dames/Project-Dames")
add_library(checkers_core "${SOURCE_DIR}/posicio.cpp" "${SOURCE_DIR}/fitxa.cpp" "${SOURCE_DIR}/tauler.cpp" "${SOURCE_DIR}/cuamoviments.cpp")
target_include_directories(checkers_core PUBLIC "${SOURCE_DIR}")
add_executable(checkers cli/main.cpp)
target_link_libraries(checkers PRIVATE checkers_core)
configure_file("${SOURCE_DIR}/tauler_inicial.txt" "${CMAKE_CURRENT_BINARY_DIR}/tauler_inicial.txt" COPYONLY)
include(CTest)
if(BUILD_TESTING)
  add_executable(checkers_tests tests/test_engine.cpp)
  target_link_libraries(checkers_tests PRIVATE checkers_core)
  add_test(NAME checkers_engine COMMAND checkers_tests)
endif()
